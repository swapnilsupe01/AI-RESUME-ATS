      def compute_cosine_similarity(v_resume, v_evidence):
      # Siamese Dense Vector Embedding Cosine Distance
      cosine_sim = np.dot(v_resume, v_evidence) / (np.linalg.norm(v_resume) * np.linalg.norm(v_evidence))
      return float(np.clip(cosine_sim, 0.0, 1.0) * 100.0)
